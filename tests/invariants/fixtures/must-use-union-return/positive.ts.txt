type Decision = Readonly<{ kind: 'accepted' }> | Readonly<{ kind: 'refused'; reason: string }>;

declare function decide(): Decision;
declare function execute(): Promise<'completed' | 'held'>;
declare function booleanOnly(): true | false;
declare const optionalBoolean: undefined | (() => boolean);
declare const condition: boolean;

void decide();
void (condition && decide());
void (condition ? decide() : decide());
void (decide(), decide());
const decision = decide();
if (decision.kind === 'refused') decision.reason;
booleanOnly();
optionalBoolean?.();

async function discardExecutionDisposition(): Promise<void> {
  void (await execute());
  const completion = await execute();
  if (completion === 'held') return;
}
