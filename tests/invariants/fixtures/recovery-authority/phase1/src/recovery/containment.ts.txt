declare const recoverySourceBrand: unique symbol;

export interface RecoverySource<Raw> {
  readonly boundary: string;
  readonly [recoverySourceBrand]: Raw;
}

type RecoverySourceDefinition<Raw> = {
  readonly boundary: string;
  scan(): readonly Raw[];
};

export const sourceDefinitions = new WeakMap<object, RecoverySourceDefinition<unknown>>();
export const receiptValues = new WeakMap<object, unknown>();

export function defineRecoverySource<Raw>(definition: RecoverySourceDefinition<Raw>): RecoverySource<Raw> {
  const source = { boundary: definition.boundary } as RecoverySource<Raw>;
  sourceDefinitions.set(source, definition as RecoverySourceDefinition<unknown>);
  return source;
}

export function defineCompositeRecoverySource<T, Raw>(
  receipts: readonly T[],
  definition: RecoverySourceDefinition<Raw>,
): RecoverySource<Raw> {
  void receipts;
  void receiptValues;
  return defineRecoverySource(definition);
}
